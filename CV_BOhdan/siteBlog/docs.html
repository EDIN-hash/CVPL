1. Обзор проекта

Автономный агрегатор-терминал для отслеживания игровых трендов, технологических новостей индустрии и официальных патчей. Разделен на изолированные потоки: социальный хайп (HN) и официальные апдейты (Steam News).

PROD Live Client Repository Link

2. Архитектура и лимиты Free Tier

  • Фронтенд: Статика на GitHub Pages (Read-Only).
  • Бэкенд: Cloudflare Workers (ES Modules, export default).
  • Хранилище: Cloudflare D1 (Serverless SQLite).
  • Ограничение: Лимит 50 внешних subrequests за запуск воркера (парсинг оптимизирован под пачки по 30–40 постов).

3. Схема базы данных (D1)

DROP TABLE IF EXISTS hype_posts;

CREATE TABLE hype_posts (
  source TEXT,
  source_type TEXT DEFAULT 'SOCIAL',
  external_id TEXT,
  title TEXT,
  url TEXT,
  score INTEGER,
  comments INTEGER,
  hype INTEGER,
  created_at INTEGER,
  first_seen_at INTEGER,
  last_seen_at INTEGER,
  PRIMARY KEY(source, external_id)
);

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type",
  "Content-Type": "application/json; charset=utf-8"
};

const CONFIG = {
  gamingKeywords: ["steam", "playstation", "xbox", "nintendo", "switch", "rtx", "radeon", "zelda", "mario", "minecraft", "cyberpunk", "valve", "riot", "blizzard", "proton", "raytracing", "dlss", "fsr", "steamdeck", "epic games", "ubisoft", "ea sports", "bethesda", "fromsoftware", "gta vi", "gta 6"],
  excludeWords: ["pdf", "arxiv", "research", "paper", "algorithm", "rendering", "topology", "siggraph", "ieee", "vector graphics", "san francisco", "city as a", "map", "geospatial", "hiring", "job", "jobs", "career", "remote", "position", "intern", "salary", "resume", "cv", "yc ", "y combinator", "saas", "startup", "crypto", "web3", "nft", "blockchain", "token", "bitcoin"],
  triggerWords: ["fail", "bug", "scandal", "leak", "worst", "ban", "update", "release"],
  hnTags: ["show hn:", "launch hn:", "ask hn:"],
  steamApps: [
    { id: 1091500, name: "Cyberpunk 2077" },

export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") return new Response(null, { status: 204, headers: corsHeaders });
    const url = new URL(request.url);

    if (url.pathname === "/force-sync") {
      try {
        await this.processFeed(env);
        return new Response(JSON.stringify({ status: "Forced scan complete" }), { headers: corsHeaders });
      } catch (e) {
        return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: corsHeaders });
      }
    }
    
    if (request.method === "GET") {
      try {
        const { results } = await env.DB.prepare("SELECT * FROM hype_posts ORDER BY hype DESC, created_at DESC LIMIT 100").all();
        return new Response(JSON.stringify(results), { headers: corsHeaders });
      } catch (err) {
        return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: corsHeaders });
      }
    }
    return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405, headers: corsHeaders });
  },

  async scheduled(event, env, ctx) { ctx.waitUntil(this.processFeed(env)); },

  async processFeed(env) {
    const now = Date.now();
    const nowSec = Math.floor(now / 1000);
    const statements = [];
    const cutoff = now - (7 * 24 * 60 * 60 * 1000);
    statements.push(env.DB.prepare("DELETE FROM hype_posts WHERE last_seen_at < ?").bind(cutoff));

    try {
      const [newRes, bestRes] = await Promise.all([
        fetch("https://hacker-news.firebaseio.com/v0/newstories.json"),
        fetch("https://hacker-news.firebaseio.com/v0/beststories.json")
      ]);
      const newIds = await newRes.json();
      const bestIds = await bestRes.json();
      const selectedIds = [...new Set([...newIds.slice(0, 25), ...bestIds.slice(0, 25)])].slice(0, 30);
      const items = await Promise.all(selectedIds.map(id => fetch(`https://hacker-news.firebaseio.com/v0/item/${id}.json`).then(r => r.ok ? r.json() : null).catch(() => null)));

      const rawPosts = items.filter(i => i && i.title).map(item => {
        const titleLower = item.title.toLowerCase();
        if (CONFIG.excludeWords.some(w => titleLower.includes(w))) return null;
        if (!CONFIG.gamingKeywords.some(w => titleLower.includes(w))) return null;
        return { id: String(item.id), title: item.title, url: item.url || `https://news.ycombinator.com/item?id=${item.id}`, score: Math.max(0, Number(item.score) || 0), comments: Math.max(0, Number(item.descendants) || 0), created: Number(item.time) || nowSec };
      }).filter(Boolean);

      if (rawPosts.length > 0) {
        const lambda = Math.LN2 / 42;
        const metrics = rawPosts.map(post => {
          const ageHours = Math.max(0.1, (nowSec - post.created) / 3600);
          const baseMetric = Math.log1p(post.score) * 18 + Math.log1p(post.comments) * 24;
          const velocity = baseMetric / Math.sqrt(Math.max(2, ageHours));
          return { ...post, ageHours, velocity };
        });
        const sortedVel = metrics.map(p => p.velocity).sort((a,b) => a-b);
        const median = sortedVel[Math.floor(sortedVel.length / 2)] || 0.1;
        const mad = sortedVel.map(v => Math.abs(v - median)).sort((a,b) => a-b)[Math.floor(sortedVel.length / 2)] || 1;

        metrics.forEach(post => {
          const robustZ = (post.velocity - median) / (mad * 1.4826);
          const hype = Math.round(100 * (1 + Math.max(0, Math.min(robustZ, 4)) * 0.4) * Math.exp(-lambda * post.ageHours));
          statements.push(env.DB.prepare(`INSERT INTO hype_posts (source, source_type, external_id, title, url, score, comments, hype, created_at, first_seen_at, last_seen_at) VALUES (?, 'SOCIAL', ?, ?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(source, external_id) DO UPDATE SET title = excluded.title, score = excluded.score, comments = excluded.comments, hype = excluded.hype, last_seen_at = excluded.last_seen_at`).bind("HN_PURE", post.id, post.title.trim(), post.url, post.score, post.comments, hype, post.created * 1000, now, now));
        });
      }
    } catch(e) {}

    try {
      for (const app of CONFIG.steamApps) {
        const res = await fetch(`https://api.steampowered.com/ISteamNews/GetNewsForApp/v2/?appid=${app.id}&count=3&maxlength=300`);
        if (!res.ok) continue;
        const data = await res.json();
        (data?.appnews?.newsitems || []).forEach(item => {
          const ageDays = (nowSec - item.date) / 86400;
          if (ageDays <= 14) {
            const hype = Math.max(20, Math.round(80 - (ageDays * 4)));
            statements.push(env.DB.prepare(`INSERT INTO hype_posts (source, source_type, external_id, title, url, score, comments, hype, created_at, first_seen_at, last_seen_at) VALUES (?, 'OFFICIAL', ?, ?, ?, 0, 0, ?, ?, ?, ?) ON CONFLICT(source, external_id) DO UPDATE SET title = excluded.title, hype = excluded.hype, last_seen_at = excluded.last_seen_at`).bind(`STEAM: ${app.name}`, String(item.gid), item.title.trim(), item.url, hype, item.date * 1000, now, now));
          }
        });
      }
    } catch(e) {}

    await env.DB.batch(statements);
  }
};

6. GitHub Actions Workflow (.github/workflows/deploy.yml)

name: Deploy GitHub Pages

on:
  push:
    branches: [ "main", "master" ]

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: "pages"
  cancel-in-progress: true

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Setup Pages
        uses: actions/configure-pages@v4
      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: '.'
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
